latest_station = None
complete_by_station = {}
failures = {}

for station_id in tqdm(STATION_IDS, desc="Preprocessing", unit="station"):
    try:
        station = preprocess_station(
            station_id,
            raw_dir=RAW_DIR,
            max_gap_hours=MAX_INTERPOLATION_GAP_HOURS,
            weather_variables=WEATHER_VARIABLES,
        )
        complete_by_station[station_id] = station
        print(f"Preprocessed {len(station):,} hourly rows for {station_id}")
        latest_station = station
    except Exception as error:  # noqa: BLE001 -- aggregate every station failure
        failures[station_id] = error
        print(f"Failed {station_id}: {error}")

if latest_station is not None:
    latest_station.info()
    display(latest_station.head())
else:
    print("No preprocessed DataFrame is available to display.")
\end{minted}

# Every station is cut at one boundary: the target station's own chronological
# split. Splitting each station at its own floor((1-test_fraction)*N) instead
# lands the boundaries on different hours, because the timelines differ in
# length, and the rows in between are dropped by the joined completeness gate.
train_by_station, test_by_station, split_boundary_utc = (
    split_station_frames_at_target_boundary(
        retained_by_station,
        target_station_id=TARGET_STATION_ID,
        test_fraction=TEST_FRACTION,
    )
)
print(f"Shared split boundary (last training hour): {split_boundary_utc.isoformat()}")

for station_id in retained_station_ids:
    write_preprocess_artifacts(
        train_by_station[station_id],
        test_by_station[station_id],
        station_id=station_id,
        output_dir=SEPARATE_DIR,
        split_boundary_utc=split_boundary_utc,
        test_fraction=TEST_FRACTION,
    )
    print(
        f"Saved {len(train_by_station[station_id]):,} train and "
        f"{len(test_by_station[station_id]):,} test rows for {station_id}"
    )
