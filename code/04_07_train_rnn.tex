\begin{minted}{python}
cv_results_rows = []
cv_horizon_rows_by_candidate = {}
sequences_by_length = {}
sampled_hyperparameters = deduplicate_sampled_parameters(
    list(
        ParameterSampler(
            RNN_PARAM_DISTRIBUTIONS, n_iter=RNN_N_ITER, random_state=RANDOM_STATE
        )
    ),
    key=sampled_parameter_key,
)
sampled_hyperparameter_keys = {
    sampled_parameter_key(params) for params in sampled_hyperparameters
}
expected_candidate_keys = {
    (subset_name, *hyperparameter_key)
    for subset_name in FEATURE_SUBSETS
    for hyperparameter_key in sampled_hyperparameter_keys
}
search_progress = tqdm(
    total=len(expected_candidate_keys) * N_VALIDATION_FOLDS,
    desc="RNN CV search",
    unit="fit",
)


def fit_predict_fold(
    train_indices, validation_indices, *, sequences, targets, architecture
):
    """Fit one candidate on a fold's training windows; predict its validation windows."""
    fold_model = build_rnn_estimator(**architecture, **RNN_FIXED_PARAMS)
    fold_model.fit(sequences[train_indices], targets[train_indices])
    return (
        pd.DataFrame(targets[validation_indices], columns=TARGET_COLUMNS),
        fold_model.predict(sequences[validation_indices]),
    )


for sequence_length in sorted({key[1] for key in sampled_hyperparameter_keys}):
    # Built over every subset's channels so all subsets share these rows and folds.
    cohort_sequences, train_targets, train_origins = build_sequences(
        train_raw_channel_frame,
        train_rows,
        sequence_length=sequence_length,
        channel_columns=FULL_CHANNEL_COLUMNS,
        target_columns=TARGET_COLUMNS,
        artifact_name="train",
    )
    cv_splits, validation_test_size = time_series_splits(
        len(train_origins),
        initial_train_fraction=INITIAL_TRAIN_FRACTION,
        n_validation_folds=N_VALIDATION_FOLDS,
        embargo_rows=EMBARGO_HOURS,
    )
    sequences_by_length[sequence_length] = {
        "cohort_sequences": cohort_sequences,
        "train_targets": train_targets,
        "train_origins": train_origins,
    }
    for subset_name, channel_columns in FEATURE_SUBSETS.items():
        train_sequences = cohort_sequences[
            :, :, subset_channel_indices(FULL_CHANNEL_COLUMNS, channel_columns)
        ]
        channel_params = {
            "channel_count": len(channel_columns),
            "channel_columns": json.dumps(channel_columns),
        }
        for candidate_key in sorted(
            key
            for key in expected_candidate_keys
            if key[0] == subset_name and key[2] == sequence_length
        ):
            candidate = dict(zip(CANDIDATE_COLUMNS, candidate_key, strict=True))
            architecture = {
                name: value for name, value in candidate.items() if name != "subset"
            }
            cv = run_cv_candidate(
                "rnn_cv_" + "_".join(map(str, candidate_key)),
                partial(
                    fit_predict_fold,
                    sequences=train_sequences,
                    targets=train_targets,
                    architecture=architecture,
                ),
                folds=cv_splits,
                execution_uuid=NOTEBOOK_EXECUTION_UUID,
                tags=candidate,
                parent_params={
                    **candidate,
                    **channel_params,
                    **SEARCH_PARAMS,
                    "search_unique_candidate_count": len(expected_candidate_keys),
                    "validation_test_size": validation_test_size,
                    "sequence_eligible_train_rows": len(train_origins),
                },
                fold_params={**candidate, **INPUT_PARQUET_SHA256_PARAMS},
                target_columns=TARGET_COLUMNS,
                station_id=station_id,
                window_timestamps=train_origins["timestamp"],
                gap_rows=EMBARGO_HOURS,
                forecast_horizon_hours=FORECAST_HORIZON_HOURS,
                progress=search_progress,
                describe_fold=lambda fold, candidate=candidate: (
                    f"RNN CV search ({candidate['subset']}, {candidate['cell_type']}, "
                    f"sequence_length={candidate['sequence_length']}, "
                    f"hidden_size={candidate['hidden_size']}, "
                    f"num_layers={candidate['num_layers']}, fold {fold})"
                ),
            )
            cv_horizon_rows_by_candidate[candidate_key] = cv.fold_horizon_rows
            cv_results_rows.append(
                cv_result_row(
                    {
                        **candidate,
                        "channel_count": len(channel_columns),
                        "dropout": RNN_DROPOUT,
                        "sequence_eligible_train_rows": len(train_origins),
                    },
                    cv.parent_metrics,
                )
            )
        del train_sequences

search_progress.close()
cv_results = verified_cv_results(
