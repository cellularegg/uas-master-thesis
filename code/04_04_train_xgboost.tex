    unit="fit",

def fit_predict_fold(
    train_indices, validation_indices, *, feature_columns, hyperparameters
):
    """Fit one candidate on a fold's training rows; predict its validation rows."""
    fold_train_rows = train_rows.iloc[train_indices]
    fold_validation_rows = train_rows.iloc[validation_indices]
    fold_model = build_xgboost_estimator(
        **hyperparameters, n_jobs=XGBOOST_N_JOBS, random_state=RANDOM_STATE
    )
    fold_model.fit(
        numeric_predictors(fold_train_rows, feature_columns),
        fold_train_rows[TARGET_COLUMNS],
    )
    return fold_validation_rows[TARGET_COLUMNS], fold_model.predict(
        numeric_predictors(fold_validation_rows, feature_columns)
    )


for (subset_name, feature_columns), params in product(
    FEATURE_SUBSETS.items(), sampled_hyperparameters
):
    hyperparameters = dict(
        zip(HYPERPARAMETER_COLUMNS, sampled_parameter_key(params), strict=True)
    )
    label = (
        f"md{hyperparameters['max_depth']}_lr{hyperparameters['learning_rate']:g}_"
        f"ne{hyperparameters['n_estimators']}_ss{hyperparameters['subsample']:g}_"
        f"cs{hyperparameters['colsample_bytree']:g}_rl{hyperparameters['reg_lambda']:g}"
    )
    candidate_params = {
        "subset": subset_name,
        "feature_count": len(feature_columns),
        **INPUT_PARQUET_SHA256_PARAMS,
        **hyperparameters,
    }
    candidate = run_cv_candidate(
        f"xgboost_cv_{subset_name}_{label}",
        partial(
            fit_predict_fold,
            feature_columns=feature_columns,
            hyperparameters=hyperparameters,
        ),
        tags={"subset": subset_name},
        parent_params={
            **candidate_params,
            "feature_columns": json.dumps(feature_columns),
            **SEARCH_PARAMS,
            "search_unique_candidate_count": len(sampled_hyperparameters),
        },
        fold_params=candidate_params,
        progress=search_progress,
        describe_fold=lambda fold, subset_name=subset_name, label=label: (
            f"XGBoost CV search ({subset_name}, {label}, fold {fold})"
        ),
        **CV_RUN_SETTINGS,
    )
    candidate_key = (subset_name, *sampled_parameter_key(params))
    cv_horizon_rows_by_candidate[candidate_key] = candidate.fold_horizon_rows
    cv_results_rows.append(
        cv_result_row(
            {

                "feature_count": len(feature_columns),
                **hyperparameters,
            },
            candidate.parent_metrics,
        )
    )

search_progress.close()
cv_results = verified_cv_results(
